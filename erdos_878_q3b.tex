\title{Disproof of Eventual Equality in Erd\H{o}s Problem 878 (Question 3b)}
\author{Ilia Pauzner}
\date{}

\begin{document}
\maketitle

For an integer $n \ge 1$, let $\mathcal{P}(n)$ denote the set of prime factors of $n$, and for a prime $p$ let $q_p(n) := p^{\lfloor \log_p n \rfloor}$ be the largest power of $p$ not exceeding $n$. Erd\H{o}s~\cite{Er84e} defined
\[
f(n) := \sum_{p \in \mathcal{P}(n)} q_p(n), \qquad F(n) := \max \sum_{a \in A} a,
\]
where $A \subseteq [2, n]$ ranges over finite sets of pairwise coprime integers $a$ with $\mathcal{P}(a) \subseteq \mathcal{P}(n)$, and asked whether the maxima $\maxUpTo f\, x := \max_{n \le x} f(n)$ and $\maxUpTo F\, x := \max_{n \le x} F(n)$ satisfy $\maxUpTo f\, x = \maxUpTo F\, x$ for all sufficiently large $x$.

\begin{theorem}
For every $X_0 \in \mathbb{N}$, there exists $x \ge X_0$ such that $\maxUpTo f\, x < \maxUpTo F\, x$.
\end{theorem}

In particular, $\maxUpTo f\, x \ne \maxUpTo F\, x$ for arbitrarily large $x$, so the answer to Question 3b is negative.

\begin{lemma}
If $p < r$ are primes with $98 r < 100 p$, then for every $n \ge p^{p^2 + 2}$ there exists an integer $z \in [2, n]$ with $\mathcal{P}(z) \subseteq \{p, r\}$ and $98 n < 100 z$.
\end{lemma}

\begin{proof}
Put $\ell = \lfloor \log_p n \rfloor$, $J = p^2 + 1 < \ell$, and $z_j = p^{\ell - j} r^j$. Then $z_0 = q_p(n) \le n$, while
\[
z_J \ge p^{\ell - J}(p + 1)^J \ge p^{\ell - J}(p^J + J p^{J - 1}) = p^{\ell - J}\bigl(p^J + (p^2 + 1)p^{J - 1}\bigr) > p^{\ell + 1} > n.
\]
Choose $0 \le j < J$ with $z_j \le n < z_{j + 1}$. Since $z_{j + 1} p = z_j r$, we have $98 n p < 98 z_{j + 1} p = 98 z_j r < 100 z_j p$, so $z = z_j$ has the required properties.
\end{proof}

Fix the six primes
\[
\begin{aligned}
q_1 &:= 168000061, & q_2 &:= 168000101, & q_3 &:= 168000121, \\
q_4 &:= 168000139, & q_5 &:= 168000143, & q_6 &:= 168000167,
\end{aligned}
\]
let $Q_6 := \{q_1, \dots, q_6\}$, and let $C_6 := \prod_{p \in Q_6} p$. Applying Lemma~1 to the pairs $(q_1, q_2)$, $(q_3, q_4)$, and $(q_5, q_6)$ (each satisfies $q_{2i-1} < q_{2i}$ and $98 q_{2i} < 100 q_{2i-1}$) yields the threshold $\Nsmooth := q_1^{q_1^2 + 2} + q_3^{q_3^2 + 2} + q_5^{q_5^2 + 2}$, which exceeds each $q_{2i-1}^{q_{2i-1}^2 + 2}$, such that for every $m \ge \Nsmooth$ there exist $b_1, b_2, b_3 \in [2, m]$ with $\mathcal{P}(b_i) \subseteq \{q_{2i-1}, q_{2i}\}$ and $100 b_i > 98 m$. Whenever $m \ge \Nsmooth$, $R' \subseteq \mathcal{P}(m)$ with $R' \cap Q_6 = \emptyset$, and $Q_6 \subseteq \mathcal{P}(m)$, the integers $b_1, b_2, b_3$ and $q_p(m) \in [2, m]$ ($p \in R'$) are pairwise coprime with prime factors dividing $m$, so
\begin{equation}\label{eq:F-lower}
100 F(m) > 294 m + 100 \sum_{p \in R'} q_p(m).
\end{equation}

\begin{lemma}
For all $n \ge 1$, $f(n) < f(2n)$.
\end{lemma}

\begin{proof}
Passing from $n$ to $2n$ never decreases $q_p$ for $p \in \mathcal{P}(n)$, and strictly increases the $p = 2$ term (either $q_2(2n) = 2 q_2(n)$ if $2 \mid n$, or $q_2(2n) \ge 2$ is adjoined if $2 \nmid n$).
\end{proof}

\begin{lemma}
For every $R \in \mathbb{N}$, there is $N_2$ such that $\maxUpTo f\, y > R y$ for all $y \ge N_2$.
\end{lemma}

\begin{proof}
Since $\sum_{p} 1/p = \infty$, choose $N$ with $\sum_{p < N} 1/p \ge 2R + 1$ and set $P = \prod_{p < N} p$ and $N_2 = 2P$. For $y \ge N_2$, let $m = P \lfloor y / P \rfloor > y / 2$. Every prime $p < N$ divides $m$ and satisfies $q_p(m) > m / p$, so
\[
\maxUpTo f\, y \ge f(m) \ge \sum_{p < N} q_p(m) \ge m \sum_{p < N} \frac{1}{p} \ge (2R + 1) m > R y. \qedhere
\]
\end{proof}

\begin{lemma}
For every $X_0$, there is $Y \ge \max(X_0, 20)$ with $14 q_p(\lfloor 21 Y / 20 \rfloor) < Y$ for all $p \in Q_6$.
\end{lemma}

\begin{proof}
Let $Y_0 = \max(X_0, 20)$. If a prime $p \in Q_6$ fails the inequality for $Y_a := Y_0 \cdot 15^{m_1}$ and $Y_b := Y_0 \cdot 15^{m_2}$ with $0 \le m_1 < m_2 \le 6$, then $Q_a := q_p(\lfloor 21 Y_a / 20 \rfloor) \ge Y_a / 14$ and $Q_b := q_p(\lfloor 21 Y_b / 20 \rfloor) \ge Y_b / 14 \ge \frac{15}{14} Y_a > \frac{21}{20} Y_a \ge Q_a$. Since $Q_a, Q_b$ are powers of $p$,
\[
\frac{21}{20} Y_b \ge Q_b \ge p Q_a \ge \frac{168000000}{14} Y_a = 12000000 Y_a,
\]
contradicting $Y_b \le 15^6 Y_a = 11390625 Y_a$ (note that $\frac{21}{20} \cdot 11390625 = 11960156.25 < 12000000$). Thus each of the $6$ primes in $Q_6$ eliminates at most one of the $7$ exponents $m \in \{0, \dots, 6\}$, and some $Y = Y_0 \cdot 15^m$ satisfies the lemma.
\end{proof}

\begin{proof}[Proof of the Theorem]
Suppose for contradiction that $\maxUpTo F\, x \le \maxUpTo f\, x$ for all $x \ge X_0$. Put $A := 2000 C_6 + 1$. By Lemma~3 with $R = 2(A + 10)$, there is $N_2$ such that $\maxUpTo f\, y > 2(A + 10) y$ for all $y \ge N_2$. Let $Y$ be given by Lemma~4 applied to $\max(X_0, N_2, 2 \Nsmooth)$, so that $Y \ge \max(X_0, N_2, 20, 2 \Nsmooth)$ and $14 q_p(\lfloor 21 Y / 20 \rfloor) < Y$ for all $p \in Q_6$.

\medskip\noindent\textbf{Step 1.}
Suppose $y \in [Y, \lfloor 21 Y / 20 \rfloor]$ and $M \ge 1$ satisfy $100 \maxUpTo f\, y \ge 105 (M + 8) Y$. Choose $n \le y$ with $f(n) = \maxUpTo f\, y$.
\begin{itemize}
\item Then $n \ge 1$, as $f(n) = \maxUpTo f\, y \ge \frac{105 (M + 8) Y}{100} > 0$. If $2n \le y$, then $f(2n) \le \maxUpTo f\, y = f(n)$, contradicting Lemma~2; hence $y < 2n$, and $n > y / 2 \ge \Nsmooth$.
\item Since $f(n) \le |\mathcal{P}(n)| \, n$ and $20 n \le 21 Y$, we have $105(M + 8) Y \le 100 |\mathcal{P}(n)| \, n \le 105 |\mathcal{P}(n)| \, Y$, so $|\mathcal{P}(n)| \ge M + 8$.
\item Removing $Q_6$ ($\le 6$ primes) and primes $< M$ ($\le M$ integers) from $\mathcal{P}(n)$ leaves at least two distinct primes $r_a, r_b \ge M$ outside $Q_6$.
\item Put $R' := (\mathcal{P}(n) \setminus Q_6) \setminus \{r_a, r_b\}$ and $D := \bigl(\prod_{p \in R'} p\bigr) C_6$. Since $r_a r_b \prod_{p \in R'} p$ divides $n$, it is at most $n \le 2Y$; as $M^2 \le r_a r_b$, we get $M^2 D \le 2 C_6 Y$, that is,
\[
D \le \left\lfloor \frac{2 C_6 Y}{M^2} \right\rfloor.
\]
\item Let $m := D(\lfloor n / D \rfloor + 1) \in (n, y + D]$. By Lemma~4, each $q \in Q_6$ satisfies $14 q_q(n) \le 14 q_q(\lfloor 21 Y / 20 \rfloor) < Y < 2n$, so $\sum_{q \in Q_6} q_q(n) < \frac{6}{7} n$. Since every prime factor of $n$ lies in $\{r_a, r_b\} \cup R' \cup Q_6$, we have $f(n) \le q_{r_a}(n) + q_{r_b}(n) + \sum_{p \in R'} q_p(n) + \sum_{q \in Q_6} q_q(n)$. Using $q_{r_a}(n), q_{r_b}(n) \le n$ and $q_p(n) \le q_p(m)$ for $p \in R'$,
\[
f(n) < 2n + \frac{6}{7} n + \sum_{p \in R'} q_p(n) \le \frac{20}{7} n + \sum_{p \in R'} q_p(m).
\]
Since $D \mid m$, we have $R' \subseteq \mathcal{P}(m)$ and $Q_6 \subseteq \mathcal{P}(m)$; also $R' \cap Q_6 = \emptyset$ and $m > n \ge \Nsmooth$. So comparing with~\eqref{eq:F-lower} gives $F(m) - f(n) > \left(\frac{294}{100} - \frac{20}{7}\right) n = \frac{29}{350} n > \frac{29}{700} y \ge \frac{29}{700} Y > \frac{Y}{25}$. Since $m \le y + D$ and $\maxUpTo F\,(y + D) \le \maxUpTo f\,(y + D)$ (as $y + D \ge Y \ge X_0$), setting $y' = y + D$ yields
\begin{equation}\label{eq:jump}
y \le y' \le y + \left\lfloor \frac{2 C_6 Y}{M^2} \right\rfloor \quad \text{and} \quad 25 \maxUpTo f\, y' \ge 25 \maxUpTo f\, y + Y.
\end{equation}
\end{itemize}

\medskip\noindent\textbf{Step 2.}
Iterating Step~1 gives: if $Y \le y$, $M \ge 1$, $100 \maxUpTo f\, y \ge 105 (M + 8) Y$, and $y + j \lfloor 2 C_6 Y / M^2 \rfloor \le \lfloor 21 Y / 20 \rfloor$, then there is $y' \ge y$ with $y' \le y + j \lfloor 2 C_6 Y / M^2 \rfloor$ and $25 \maxUpTo f\, y' \ge 25 \maxUpTo f\, y + j Y$. This follows by induction on $j$ (for $j = 0$ take $y' = y$), since the $y'$ obtained for $j$ satisfies $Y \le y' \le \lfloor 21 Y / 20 \rfloor$ and $\maxUpTo f\, y' \ge \maxUpTo f\, y$, so that Step~1 applies to it.

Let $\rem(k) := \frac{54 C_6 Y}{A + k}$, and note that $20(z + \rem(k)) \le 21 Y$ implies $z \le \lfloor 21 Y / 20 \rfloor$, since $\rem(k) \ge 0$ and $z \in \mathbb{N}$. We construct $y_k \ge Y$ by induction on $k \ge 0$ with
\[
20(y_k + \rem(k)) \le 21 Y \quad \text{and} \quad 100 \maxUpTo f\, y_k \ge 105(A + 1 + k + 8) Y.
\]
For $k = 0$, $y_0 = Y$ satisfies both inequalities since $20 \rem(0) = \frac{1080 C_6 Y}{2000 C_6 + 1} < Y$ and $100 \maxUpTo f\, Y > 200 (A + 10) Y \ge 105 (A + 1 + 0 + 8) Y$. Given $y_k$, set $M = A + 1 + k \ge 1$ and $\step = \lfloor 2 C_6 Y / M^2 \rfloor$. Since $\step \cdot M^2 \le 2 C_6 Y$ and $(A + k)(A + k + 1) \le M^2$,
\[
27 \cdot \step \le \frac{54 C_6 Y}{(A + k)(A + k + 1)} = \rem(k) - \rem(k + 1),
\]
so $20(y_k + 27 \cdot \step + \rem(k + 1)) \le 20(y_k + \rem(k)) \le 21 Y$ and hence $y_k + 27 \cdot \step \le \lfloor 21 Y / 20 \rfloor$. Iterating Step~1 $27$ times from $y_k$ (with this $M$, for which $105(M + 8) Y = 105(A + 1 + k + 8) Y$) therefore gives $y_{k + 1}$ with $y_k \le y_{k + 1} \le y_k + 27 \cdot \step$, so that $20(y_{k + 1} + \rem(k + 1)) \le 21 Y$, and $100 \maxUpTo f\, y_{k + 1} \ge 100 \maxUpTo f\, y_k + 27 \cdot 4 Y \ge 105(A + 1 + k + 8) Y + 108 Y > 105(A + 1 + (k + 1) + 8) Y$.

Taking $K = \maxUpTo f\, \lfloor 21 Y / 20 \rfloor + 1$, the first inequality for $y_K$ gives $y_K \le \lfloor 21 Y / 20 \rfloor$, hence $\maxUpTo f\, y_K \le \maxUpTo f\, \lfloor 21 Y / 20 \rfloor < K$, while the second gives $100 \maxUpTo f\, y_K \ge 105(A + 1 + K + 8) Y > 100 K$, a contradiction.
\end{proof}
